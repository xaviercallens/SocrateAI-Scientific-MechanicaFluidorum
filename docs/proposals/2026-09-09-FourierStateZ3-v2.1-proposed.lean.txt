/-! ### 1. The wavevector lattice and norm -/

/-- Wavevectors in ℤ³. `abbrev` so the additive group structure flows through. -/
abbrev Wavevector := Fin 3 → ℤ

/-- Squared magnitude |k|² ∈ ℤ. -/
def k_sq (k : Wavevector) : ℤ := ∑ i : Fin 3, (k i) ^ 2

theorem k_sq_nonneg (k : Wavevector) : 0 ≤ k_sq k :=
  Finset.sum_nonneg fun _ _ => sq_nonneg _

/-- The load-bearing lemma: |k|² = 0 iff k = 0. Lets every `k_sq = 0` branch collapse safely. -/
theorem k_sq_eq_zero_iff (k : Wavevector) : k_sq k = 0 ↔ k = 0 := by
  constructor
  · intro h
    ext i
    have h1 : ∀ j ∈ Finset.univ, (0 : ℤ) ≤ (k j) ^ 2 := fun j _ => sq_nonneg (k j)
    have h2 := (Finset.sum_eq_zero_iff_of_nonneg h1).mp h i (Finset.mem_univ i)
    exact sq_eq_zero_iff.mp h2
  · rintro rfl
    simp [k_sq]

theorem k_sq_neg (k : Wavevector) : k_sq (-k) = k_sq k := by
  simp [k_sq, neg_sq]

/-! ### 2. Complex dot product -/

noncomputable def fourier_dot (k : Wavevector) (v : Fin 3 → ℂ) : ℂ :=
  ∑ i : Fin 3, (k i : ℂ) * v i

/-! ### 3. The constrained kinematic state space -/

/-- A physically valid Fourier state: divergence-free, conjugate symmetry for real fields, zero mean. -/
structure FourierState where
  u : Wavevector → (Fin 3 → ℂ)
  div_free : ∀ k : Wavevector, fourier_dot k (u k) = 0
  conj_sym : ∀ k : Wavevector, ∀ i : Fin 3, u (-k) i = conj (u k i)
  zero_mean : ∀ i : Fin 3, u 0 i = 0

/-! ### 4. Witnesses -/

noncomputable def zeroState : FourierState where
  u := fun _ _ => 0
  div_free := fun k => by simp [fourier_dot]
  conj_sym := fun _ _ => by simp
  zero_mean := fun _ => rfl

def k0 : Wavevector := ![1, 0, 0]
noncomputable def modeAmp : Fin 3 → ℂ := ![0, 1, 0]

theorem k0_ne_zero : k0 ≠ 0 := by
  intro h
  have h0 := congrFun h 0
  simp [k0] at h0

theorem k0_ne_neg : k0 ≠ -k0 := by
  intro h
  have h0 := congrFun h 0
  simp [k0] at h0

theorem k_sq_k0 : k_sq k0 = 1 := by
  norm_num [k_sq, k0, Fin.sum_univ_three]

/-- A state with transverse real amplitude. -/
noncomputable def pairState : Wavevector → Fin 3 → ℂ :=
  fun k => if k = k0 then modeAmp else if k = -k0 then modeAmp else 0

theorem pairState_div_free : ∀ k : Wavevector, fourier_dot k (pairState k) = 0 := by
  intro k
  unfold pairState fourier_dot
  split_ifs with h1 h2
  · norm_num [k0, modeAmp, Fin.sum_univ_three]
  · norm_num [k0, modeAmp, Fin.sum_univ_three]
  · simp

theorem modeAmp_real (i : Fin 3) : conj (modeAmp i) = modeAmp i := by
  fin_cases i <;> simp [modeAmp]

theorem pairState_conj_sym : ∀ (k : Wavevector) (i : Fin 3), pairState (-k) i = conj (pairState k i) := by
  intro k i
  unfold pairState
  by_cases h1 : k = k0
  · subst h1
    rw [if_neg k0_ne_neg.symm, if_pos rfl, if_pos rfl, modeAmp_real]
  · by_cases h2 : k = -k0
    · subst h2
      have h3 : - -k0 = k0 := neg_neg k0
      rw [h3, if_pos rfl, if_neg k0_ne_neg, if_pos rfl, modeAmp_real]
    · have hn1 : -k ≠ k0 := fun h => h2 (neg_eq_iff_eq_neg.mp h)
      have hn2 : -k ≠ -k0 := fun h => h1 (neg_injective h)
      rw [if_neg hn1, if_neg hn2, if_neg h1, if_neg h2, map_zero]

/-- The nontrivial witness: constraints admit genuinely nonzero states. -/
noncomputable def pairWitness : FourierState where
  u := pairState
  div_free := pairState_div_free
  conj_sym := pairState_conj_sym
  zero_mean := fun _ => by
    unfold pairState
    have hz1 : (0 : Wavevector) ≠ k0 := k0_ne_zero.symm
    have hz2 : (0 : Wavevector) ≠ -k0 := fun h => k0_ne_zero (neg_eq_zero.mp h.symm)
    rw [if_neg hz1, if_neg hz2]

theorem pairWitness_nontrivial : pairWitness.u k0 1 = 1 := by
  show pairState k0 1 = 1
  unfold pairState
  rw [if_pos rfl]
  rfl

/-! ### 5. The Galerkin (finite) variant -/

structure GalerkinState (M : ℕ) extends FourierState where
  cutoff : ∀ k : Wavevector, (M : ℤ) ^ 2 < k_sq k → toFourierState.u k = 0

noncomputable def pairGalerkin : GalerkinState 1 where
  toFourierState := pairWitness
  cutoff := by
    intro k hk
    show pairState k = 0
    unfold pairState
    split_ifs with h1 h2
    · subst h1; rw [k_sq_k0] at hk; norm_num at hk
    · subst h2; rw [k_sq_neg, k_sq_k0] at hk; norm_num at hk
    · rfl

/-! ### 6. The Leray projector (ℂ-valued, Indicator-Sum Robust) -/

noncomputable def LerayProjector (k : Wavevector) (i j : Fin 3) : ℂ :=
  if k_sq k = 0 then (if i = j then 1 else 0)
  else (if i = j then 1 else 0) - (k i : ℂ) * (k j : ℂ) / (k_sq k : ℂ)

theorem leray_conj (k : Wavevector) (i j : Fin 3) :
    conj (LerayProjector k i j) = LerayProjector k i j := by
  unfold LerayProjector; split_ifs <;> simp

theorem leray_symm (k : Wavevector) (i j : Fin 3) :
    LerayProjector k i j = LerayProjector k j i := by
  unfold LerayProjector
  have hδ : (if i = j then (1 : ℂ) else 0) = (if j = i then (1 : ℂ) else 0) := by
    by_cases hij : i = j
    · rw [if_pos hij, if_pos hij.symm]
    · rw [if_neg hij, if_neg (fun h => hij h.symm)]
  split_ifs with h
  · exact hδ
  · rw [hδ]; ring

theorem leray_col_orthogonal (k : Wavevector) (j : Fin 3) :
    ∑ i : Fin 3, (k i : ℂ) * LerayProjector k i j = 0 := by
  by_cases h : k_sq k = 0
  · have hk := (k_sq_eq_zero_iff k).mp h
    subst hk
    simp
  · have hS : (k_sq k : ℂ) ≠ 0 := Int.cast_ne_zero.mpr h
    unfold LerayProjector
    simp only [if_neg h]
    have hcast3 : (k_sq k : ℂ) = ∑ i : Fin 3, (k i : ℂ) ^ 2 := by
      unfold k_sq
      push_cast
      exact Finset.sum_congr rfl (fun x _ => rfl)
    calc
      ∑ i : Fin 3, (k i : ℂ) * ((if i = j then 1 else 0) - (k i : ℂ) * (k j : ℂ) / (k_sq k : ℂ))
      _ = ∑ i : Fin 3, ((k i : ℂ) * (if i = j then 1 else 0) - (k i : ℂ) * (k i : ℂ) * (k j : ℂ) / (k_sq k : ℂ)) := by
        refine Finset.sum_congr rfl fun i _ => ?_
        ring
      _ = (∑ i : Fin 3, (k i : ℂ) * (if i = j then 1 else 0)) - (∑ i : Fin 3, (k i : ℂ) ^ 2 * (k j : ℂ) / (k_sq k : ℂ)) := by
        rw [Finset.sum_sub_distrib]
        refine congr_arg₂ _ rfl (Finset.sum_congr rfl fun i _ => ?_)
        ring
      _ = (k j : ℂ) * 1 - (∑ i : Fin 3, (k i : ℂ) ^ 2) * (k j : ℂ) / (k_sq k : ℂ) := by
        congr 1
        · exact Finset.sum_ite_eq' Finset.univ j (fun i => (k i : ℂ)) (Finset.mem_univ j)
        · rw [← Finset.sum_mul]
          exact Finset.sum_congr rfl fun i _ => rfl
      _ = 0 := by
        rw [← hcast3, mul_one, mul_div_cancel_right₀ _ hS, sub_self]

/-! ### 7. The projector as an operator -/

noncomputable def applyLeray (k : Wavevector) (v : Fin 3 → ℂ) : Fin 3 → ℂ :=
  fun i => ∑ j : Fin 3, LerayProjector k i j * v j

theorem applyLeray_zero (k : Wavevector) :
    applyLeray k (0 : Fin 3 → ℂ) = 0 := by
  ext i; simp [applyLeray]

/-- DoD-1 (Transversality). k annihilates the projected field. -/
theorem applyLeray_div_free (k : Wavevector) (v : Fin 3 → ℂ) :
    fourier_dot k (applyLeray k v) = 0 := by
  unfold fourier_dot applyLeray
  calc ∑ i : Fin 3, (k i : ℂ) * ∑ j : Fin 3, LerayProjector k i j * v j
      = ∑ i : Fin 3, ∑ j : Fin 3, (k i : ℂ) * LerayProjector k i j * v j := by
        refine Finset.sum_congr rfl fun i _ => ?_
        rw [Finset.mul_sum]
        exact Finset.sum_congr rfl fun j _ => by ring
    _ = ∑ j : Fin 3, ∑ i : Fin 3, (k i : ℂ) * LerayProjector k i j * v j := Finset.sum_comm
    _ = ∑ j : Fin 3, (∑ i : Fin 3, (k i : ℂ) * LerayProjector k i j) * v j := by
        refine Finset.sum_congr rfl fun j _ => ?_
        rw [← Finset.sum_mul]
    _ = 0 := by
        refine Finset.sum_eq_zero fun j _ => ?_
        rw [leray_col_orthogonal k j, zero_mul]

/-- DoD-2 (Fixed point). Transverse fields are untouched. -/
theorem applyLeray_eq_self {k : Wavevector} {v : Fin 3 → ℂ}
    (hv : fourier_dot k v = 0) : applyLeray k v = v := by
  ext i
  unfold applyLeray LerayProjector
  by_cases h : k_sq k = 0
  · simp only [if_pos h]
    exact Finset.sum_ite_eq' Finset.univ i v (Finset.mem_univ i)
  · have hS : (k_sq k : ℂ) ≠ 0 := Int.cast_ne_zero.mpr h
    simp only [if_neg h]
    calc
      ∑ j : Fin 3, ((if i = j then (1 : ℂ) else 0) - (k i : ℂ) * (k j : ℂ) / (k_sq k : ℂ)) * v j
      _ = ∑ j : Fin 3, ((if i = j then (1 : ℂ) else 0) * v j - (k i : ℂ) * ((k j : ℂ) * v j) / (k_sq k : ℂ)) := by
        refine Finset.sum_congr rfl fun j _ => ?_
        ring
      _ = (∑ j : Fin 3, (if i = j then (1 : ℂ) else 0) * v j) - (∑ j : Fin 3, (k i : ℂ) * ((k j : ℂ) * v j) / (k_sq k : ℂ)) := by
        exact Finset.sum_sub_distrib
      _ = 1 * v i - (k i : ℂ) * (∑ j : Fin 3, (k j : ℂ) * v j) / (k_sq k : ℂ) := by
        congr 1
        · exact Finset.sum_ite_eq' Finset.univ i v (Finset.mem_univ i)
        · rw [← Finset.mul_sum]
          exact Finset.sum_congr rfl fun j _ => rfl
      _ = v i := by
        have h_dot := hv
        unfold fourier_dot at h_dot
        rw [h_dot, mul_zero, zero_div, sub_zero, one_mul]

/-- DoD-3 (Idempotence, Deviation D2). One line proof. -/
theorem applyLeray_idem (k : Wavevector) (v : Fin 3 → ℂ) :
    applyLeray k (applyLeray k v) = applyLeray k v :=
  applyLeray_eq_self (applyLeray_div_free k v)

/-! ### 8. Sublattice Invariance (The 2D3C Regularizing Core) -/

/-- A plane through the origin as an additive subgroup of ℤ³ (the 2D3C support). -/
def planeSubgroup (e : Fin 3 → ℤ) : AddSubgroup (Fin 3 → ℤ) where
  carrier := {k | ∑ i : Fin 3, e i * k i = 0}
  zero_mem' := by simp
  add_mem' := by
    intro a b ha hb
    simp only [Set.mem_setOf_eq] at ha hb ⊢
    simp only [Pi.add_apply, mul_add, Finset.sum_add_distrib, ha, hb, add_zero]
  neg_mem' := by
    intro a ha
    simp only [Set.mem_setOf_eq] at ha ⊢
    simp only [Pi.neg_apply, mul_neg, Finset.sum_neg_distrib, ha, neg_zero]

/-- Sublattice invariance: Any bilinear interaction drawing only on triads
    p + q = k cannot create modes outside an additive subgroup Λ. -/
theorem sublattice_invariance
    (Λ : AddSubgroup (Fin 3 → ℤ))
    (B : (Wavevector → Fin 3 → ℂ) → (Wavevector → Fin 3 → ℂ) → (Wavevector → Fin 3 → ℂ))
    (htriad : ∀ u v k, (∀ p q : Wavevector, p + q = k → u p = 0 ∨ v q = 0) → B u v k = 0)
    (u v : Wavevector → Fin 3 → ℂ)
    (hu : ∀ p, p ∉ Λ → u p = 0) (hv : ∀ q, q ∉ Λ → v q = 0)
    {k : Wavevector} (hk : k ∉ Λ) :
    B u v k = 0 := by
  apply htriad
  intro p q hpq
  by_cases hp : p ∈ Λ
  · right
    apply hv
    intro hq
    exact hk (hpq ▸ Λ.add_mem hp hq)
  · left
    exact hu p hp

theorem leray_support (Λ : AddSubgroup (Fin 3 → ℤ))
    (u : Wavevector → Fin 3 → ℂ) (hu : ∀ p, p ∉ Λ → u p = 0) :
    ∀ p, p ∉ Λ → applyLeray p (u p) = 0 := fun p hp => by
  rw [hu p hp, applyLeray_zero]
